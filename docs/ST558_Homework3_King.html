Task 4: Writing a Function

Writing getIQR Function

getIQR <- function(x,...){
  as.numeric(quantile(x,.75)-quantile(x,.25))
}

Writing getCoefVar Function

getCoefVar <- function(x,...){
  sd(x)/abs(mean(x))
}

Writing getSummaries Function

getSummaries <- function(x){
  IQR <- getIQR(x)
  CV <- getCoefVar(x)
  return(list(IQR = IQR,CoefVar = CV))
}
