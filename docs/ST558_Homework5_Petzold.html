 get_news <- function(subject, date, api_key) {
   
   url <- paste0("https://newsapi.org/v2/everything?q=", subject,
                 "&from=", date,
                 "&language=en",
                 "&apiKey=", api_key)
   
   result <- GET(url)
   parsed <- content(result, as = "text", encoding = "UTF-8") |>
     fromJSON()
   
   return(parsed$articles)
   
 }
